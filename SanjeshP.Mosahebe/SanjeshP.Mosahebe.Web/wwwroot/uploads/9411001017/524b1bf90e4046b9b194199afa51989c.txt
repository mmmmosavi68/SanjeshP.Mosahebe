    {
        [Key]
        public TKey Id { get; set; }

        [Required(ErrorMessage = "وضعیت حذف الزامی است")]

        [Display(Name = "شناسه سازنده")]
        [ForeignKey("Creator")]
        public Guid CreatedBy { get; set; }

        [Display(Name = "آی‌پی میزبان")]
        public string HostIp { get; set; }
    }

        try
        {
            // پیمایش بر روی تمام فرم‌های ارسال شده
            foreach (var formInput in Forms)

                if (cancellationToken.IsCancellationRequested) break; // توقف در صورت کنسل شدن عملیات

    $(function () {

        // Helper: اعتبارسنجی ورودی عددی
        // ===========================
        function validateScoreInput($input) {
            if (!$input || $input.length === 0) return true;

            var id = $input.attr('id') || '';
            var idx = id.replace('scoreValue_', '');
            var $feedback = $('#scoreFeedback_' + idx);

            if ($input.prop('disabled')) {
                $input.removeClass('is-invalid');
                if ($feedback.length) $feedback.hide();
                return true;
            }

            var val = $input.val();
            if (val === '' || isNaN(parseFloat(val))) {
                $input.addClass('is-invalid');
                if ($feedback.length) $feedback.text('لطفاً یک عدد معتبر وارد کنید.').show();
                return false;
            }

            var num = parseFloat(val);
            var min = parseFloat($input.attr('min') || '0');
            var maxAttr = parseFloat($input.attr('max') || '');
            var dataMax = parseFloat($input.data('one-max') || $input.data('oneMax') || '');
            var max = (!isNaN(maxAttr) && maxAttr > 0) ? maxAttr : (!isNaN(dataMax) ? dataMax : 0);

            if (num < min || (max > 0 && num > max)) {
                $input.addClass('is-invalid');
                if ($feedback.length) $feedback.text('مقدار باید بین ' + min + ' و ' + max + ' باشد.').show();
                return false;
            }

            $input.removeClass('is-invalid');
            if ($feedback.length) $feedback.hide();
            return true;
        }

        // ===========================
        // Helper: فعال/غیرفعال کردن score و rejectReason بر اساس checkbox
        // ===========================
        function toggleScoreByCheckbox(id, checked) {
            var $reject = $('#rejectReason_' + id);
            var $score = $('#scoreValue_' + id);
            var $feedback = $('#scoreFeedback_' + id);

            if (checked) {
                if ($reject.length) {
                    $reject.val('').prop('disabled', true).prop('required', false).removeClass('is-invalid');
                }
                if ($score.length) {
                    $score.prop('disabled', false).prop('required', true);
                    var dataOneMax = $score.data('oneMax') || $score.data('one-max');
                    if (!$score.attr('max') && dataOneMax) $score.attr('max', dataOneMax);
                    validateScoreInput($score);
                }
                if ($feedback.length) $feedback.hide();
            } else {
                if ($score.length) $score.prop('disabled', true).prop('required', false).val('').removeClass('is-invalid');
                if ($reject.length) $reject.prop('disabled', false).prop('required', true);
                if ($feedback.length) $feedback.hide();
            }
        }

            var $this = $(this);
            var val = $this.val();
            var m = val.match(/^\d+(\.\d{0,2})?/);
            var newVal = m ? m[0] : '';
            if (newVal !== val) $this.val(newVal);

            validateScoreInput($this);
        });

        public async Task<StudentFormSubmissionScore> GetFormSubmissionDataByStudentAndSubmissionScoresIDAsync(int studentId, int SubmissionId, CancellationToken cancellationToken)
        {
            var result = await TableNoTracking
                 .Include(s => s.FormItem)
                     .ThenInclude(f => f.FormItemDefinitions)
                 .Include(s => s.StudentFormSubmissionDatas)
                     .ThenInclude(d => d.FormItemDefinition)
                 .Include(s => s.StudentFormSubmissionDatas)
                     .ThenInclude(d => d.FormItemCoefficientDefinition)
                 .Where(s => s.StudentId == studentId && s.Id==SubmissionId && s.IsActive && !s.IsDeleted)
                   .FirstOrDefaultAsync(cancellationToken)
                   .ConfigureAwait(false);




            // همون منطق اضافه‌کردن فیلدهای خالی و مرتب‌سازی بمونه ✅

            result.StudentFormSubmissionDatas = result.StudentFormSubmissionDatas
                .Where(d => d.IsActive && !d.IsDeleted)
                .ToList();

            var allDefs = result.FormItem?.FormItemDefinitions?.Where(d => d.IsActive && !d.IsDeleted).ToList() ?? new();

            foreach (var def in allDefs)
            {
                if (!result.StudentFormSubmissionDatas.Any(d => d.FormItemDefinitionId == def.Id))
                {
                    result.StudentFormSubmissionDatas.Add(new StudentFormSubmissionData
                    {
                        FormItemDefinitionId = def.Id,
                        FormItemDefinition = def,
                        Value = null,
                        InputType = def.inputType,
                        IsActive = true
                    });
                }
            }

            result.StudentFormSubmissionDatas = result.StudentFormSubmissionDatas
                    .OrderBy(d =>d.CreatedDate)
                    .ToList();


            return result;

        }

        public async Task<IEnumerable<StudentFormSubmissionScore>> GetAllByStudentIdAsync(int studentId, CancellationToken cancellationToken)
        {
            var results = await TableNoTracking
                .Include(s => s.FormItem)
                    .ThenInclude(f => f.FormItemCoefficient)
                .Include(s => s.FormItem)
                    .ThenInclude(f => f.FormItemDefinitions)
                .Include(s => s.StudentFormSubmissionDatas)
                    .ThenInclude(d => d.FormItemDefinition)
                .Include(s => s.StudentFormSubmissionDatas)
                    .ThenInclude(d => d.FormItemCoefficientDefinition)
                .Where(s => s.StudentId == studentId && s.IsActive && !s.IsDeleted)
                .OrderByDescending(s => s.CreatedDate)
                .ToListAsync(cancellationToken);

            // همون منطق اضافه‌کردن فیلدهای خالی و مرتب‌سازی بمونه ✅
            foreach (var score in results)
            {
                var datas = score.StudentFormSubmissionDatas ?? new List<StudentFormSubmissionData>();

                score.StudentFormSubmissionDatas = score.StudentFormSubmissionDatas
                    .Where(d => d.IsActive && !d.IsDeleted)
                    .ToList();

                score.StudentFormSubmissionDatas = datas
                                                    .Where(d => d.IsActive && !d.IsDeleted)
                                                    .ToList();


                var allDefs = score.FormItem?.FormItemDefinitions?.Where(d => d.IsActive && !d.IsDeleted).ToList() ?? new();

                foreach (var def in allDefs)
                {
                    if (!score.StudentFormSubmissionDatas.Any(d => d.FormItemDefinitionId == def.Id))
                    {
                        score.StudentFormSubmissionDatas.Add(new StudentFormSubmissionData
                        {
                            FormItemDefinitionId = def.Id,
                            FormItemDefinition = def,
                            Value = null,
                            InputType = def.inputType,
                            IsActive = true
                        });
                    }
                }

                score.StudentFormSubmissionDatas = score.StudentFormSubmissionDatas
                    .OrderBy(d => d.FormItemDefinition?.CreatedDate)
                    .ToList();
            }

            return results;
        }
